%% How to plot a graph: https://tex.stackexchange.com/questions/156964/guide-to-draw-charts-basic-pie-bar-from-data
\pgfplotstableread[row sep=\\,col sep=&]{
interations & Zero & Mem100 & Mem1000 & Mem10000 & Auto  \\
complex JSON & 18547 & 1856  &  1100   &   1064    &  979  \\
complex Proto & 16644 & 1091  &  508   &  342    &  261 \\
simple Proto & 4391 & 449  & 239   & 221    & 154  \\
}\mydata
  \begin{tikzpicture}
    \begin{axis}[
      ybar=5pt,
      bar width=18pt,
      width=.98\textwidth,
      height=.6\textwidth,
      legend style={
        at={(0.5,1)},
        anchor=north,
        legend columns=-1
      },
      scaled ticks=false,
      symbolic x coords={complex JSON, complex Proto, simple Proto},
      xtick=data,
      %% These lines add text above the bars:
      nodes near coords,
      % nodes near coords align={vertical},
      %% This adds padding on the left and right, see https://tex.stackexchange.com/questions/206996/spacing-bars-in-ybar
      enlarge x limits={abs=2.1cm},
      ymin=0,
      ymax=20500,
      ylabel={nanoseconds/validation},
    ]
    %% TODO: Add back colours to bars
    \addplot[blue,fill] table[x=interations,y=Zero]{\mydata};
    \addplot[red,fill] table[x=interations,y=Mem100]{\mydata};
    \addplot[yellow,fill] table[x=interations,y=Mem1000]{\mydata};
    \addplot[orange,fill] table[x=interations,y=Mem10000]{\mydata};
    \addplot[purple,fill] table[x=interations,y=Auto]{\mydata};
    \legend{0, 100, 1000, 10000, infinity}
    \end{axis}
  \end{tikzpicture}
